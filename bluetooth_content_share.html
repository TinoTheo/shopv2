        // 5️⃣ PROFIT CALCULATION HARDENING & 7️⃣ TOTALS UPDATE
        function finalizeSale() {
            if (!isAppUnlocked()) { showToast("License required to complete sale"); return; }
            if (!validateCheckout()) { return; }
            document.getElementById('btn-finalize').disabled = true;

            // Hardened Profit Calculation
            let saleTotal = 0;
            let saleCost = 0;
            let saleProfit = 0;

            const itemsWithVoidFlag = state.cart.map(item => ({
                ...item,
                voided: false
            }));

            itemsWithVoidFlag.forEach(item => {
                const itemTotal = item.price * item.qty;
                const itemCost = item.cost * item.qty;
                const itemProfit = itemTotal - itemCost; // Deterministic math
               
                saleTotal += itemTotal;
                saleCost += itemCost;
                saleProfit += itemProfit;
            });

            // 10️⃣ Transaction ID Uniqueness Check (Prevent double count)
            const txId = generateId();
            if (state.sales.some(s => s.id === txId)) {
                showToast("Error: Duplicate Transaction ID");
                document.getElementById('btn-finalize').disabled = false;
                return;
            }

            const saleRecord = {
                id: txId,
                timestamp: new Date().toISOString(), // UTC ISO string
                items: itemsWithVoidFlag,
                total: saleTotal,
                cost: saleCost,
                profit: saleProfit, // Store profit at confirmation
                method: state.selectedPaymentMethod,
                voided: false
            };

            // Inventory Update (After Validation)
            state.cart.forEach(cartItem => {
                const product = state.products.find(p => p.id == cartItem.id);
                if(product) product.stock -= cartItem.qty;
            });

            // Update State
            state.sales.push(saleRecord);
           
            // Update Gross Totals
            state.totals.gross += saleTotal;
            state.totals.profit += saleProfit;
            state.totals.transactions += 1;

            saveState();
            state.cart = []; // Reset Cart
            updateCartUI();
            closeModal('modal-checkout');
            renderPOS();
            renderInventory(); // Update inventory to reflect new stock
           
            // Show Feedback
            showSuccessModal();
        }

        // --- BACKUP & RESTORE (CryptoJS + Compatibility) ---
        async function triggerBackup() {
            try {
                const backupPayload = {
                    version: "1.5.0",
                    deviceId: getDeviceID(),
                    inventory: state.products,
                    sales: state.sales,
                    // 8️⃣ Backward Compatibility for new fields
                    totals: state.totals,
                    settings: state.settings,
                    exportedAt: Date.now()
                };

                const payloadString = JSON.stringify(backupPayload);
                const signature = digestSHA256(payloadString + SECRET);

                const signedPayload = { ...backupPayload, signature };
                const signedPayloadString = JSON.stringify(signedPayload);

                const encryptedData = encryptData(signedPayloadString);

                const blob = new Blob([encryptedData], { type: "application/octet-stream" });
                const url = URL.createObjectURL(blob);
                const a = document.createElement("a");
                a.href = url;
                a.download = `shopmate-backup-${Date.now()}.smb`;
                a.click();
                URL.revokeObjectURL(url);

                localStorage.setItem("lastBackupDate", new Date().toDateString());
                showToast("Encrypted Backup downloaded");
            } catch (err) {
                console.error(err);
                alert("Backup failed: " + err.message);
            }
        }

        async function restoreBackup() {
            const fileInput = document.getElementById("restoreFile");
            const file = fileInput.files[0];
            if (!file) return alert("Select a backup file (.smb).");

            const reader = new FileReader();
            reader.onload = async function(e) {
                try {
                    const decryptedString = decryptData(e.target.result);
                    if (!decryptedString) throw new Error("Decryption returned empty.");

                    const signedPayload = JSON.parse(decryptedString);

                    const { signature, ...payloadWithoutSignature } = signedPayload;
                    const expectedSignature = digestSHA256(JSON.stringify(payloadWithoutSignature) + SECRET);

                    if (signature !== expectedSignature) {
                        return alert("Security Alert: Backup signature mismatch.");
                    }

                    const confirmRestore = confirm(
                        "Valid Backup Detected.\nRestore ID: " + signedPayload.deviceId + "\nThis will overwrite your current data. Continue?"
                    );

                    if (!confirmRestore) return;

                    // 9️⃣ Data Safety Guards - Default missing fields
                    state.products = Array.isArray(signedPayload.inventory) ? signedPayload.inventory : defaultProducts;
                    state.sales = Array.isArray(signedPayload.sales) ? signedPayload.sales : [];
                    state.settings = signedPayload.settings || state.settings;
                    state.totals = signedPayload.totals || { gross: 0, profit: 0, transactions: 0 };

                    saveState();

                    // Refresh UI
                    document.getElementById('setting-business-name').value = state.settings.name;
                    document.getElementById('setting-currency').value = state.settings.currency;
                    document.getElementById('business-name-display').innerText = state.settings.name;
                   
                    renderInventory();
                    renderPOS();
                    renderReports();
                   
                    alert("Restore successful. Reloading...");
                    location.reload();

                } catch (err) {
                    alert("Restore failed: Invalid file or corrupted data.");
                    console.error(err);
                }
            };
            reader.readAsText(file);
        }

        // --- 6️⃣ DAILY REPORT IMMUTABILITY ---
        function renderReports() {
            const today = new Date().toDateString();
            // Filter strictly by timestamp date string (UTC to local string is consistent in toDateString)
            const todaysSales = state.sales.filter(s => new Date(s.timestamp).toDateString() === today);
            const unlocked = isAppUnlocked();
          
            let dailySalesTotal = 0;
            let dailyProfit = 0;
            let cashTotal = 0;
            let ecoTotal = 0;
            let txCount = 0;

            todaysSales.forEach(s => {
                if (s.voided) return;
               
                // 5️⃣ Use stored profit to prevent drift
                if (s.profit !== undefined) {
                    dailySalesTotal += s.total;
                    dailyProfit += s.profit;
                    txCount++;
                } else {
                    // Fallback for old transactions without stored profit
                    let sProfit = 0;
                    s.items.forEach(i => {
                        if (typeof i.voided === 'undefined') i.voided = false;
                        if (!i.voided) {
                            const itemTotal = i.price * i.qty;
                            const itemCost = i.cost * i.qty;
                            dailySalesTotal += itemTotal;
                            sProfit += (itemTotal - itemCost);
                            txCount++;
                        }
                    });
                    dailyProfit += sProfit;
                }

                if(s.method === 'Cash') cashTotal += s.total;
                else ecoTotal += s.total;
            });

            if (unlocked) {
                document.getElementById('report-total-sales').innerText = formatMoney(dailySalesTotal);
                document.getElementById('report-total-profit').innerText = formatMoney(dailyProfit);
                document.getElementById('report-cash').innerText = formatMoney(cashTotal);
                document.getElementById('report-ecocash').innerText = formatMoney(ecoTotal);
            } else {
                document.getElementById('report-total-sales').innerText = "---";
                document.getElementById('report-total-profit').innerText = "---";
                document.getElementById('report-cash').innerText = "---";
                document.getElementById('report-ecocash').innerText = "---";
            }

            const list = document.getElementById('transaction-list');
            list.innerHTML = '';
            const recent = [...todaysSales].reverse();

            if(recent.length === 0) {
                list.innerHTML = '<div style="text-align:center; color:gray;">No sales yet today.</div>';
            } else {
                recent.forEach(s => {
                    const time = new Date(s.timestamp).toLocaleTimeString([], {hour:'2-digit', minute:'2-digit'});
                  
                    let netTotal = 0;
                    let hasVoidedItems = false;
                    s.items.forEach(i => {
                        if (typeof i.voided === 'undefined') i.voided = false;
                        if (!i.voided) netTotal += (i.price * i.qty);
                        else hasVoidedItems = true;
                    });

                    const isFullyVoided = s.voided;
                    const rowClass = isFullyVoided ? 'inv-item voided transaction-row' : 'inv-item transaction-row';
                    const itemSummary = getTransactionSummary(s.items);
                  
                    const voidAction = isFullyVoided
                        ? `<span class="text-gray text-small" style="margin-left: auto;">Voided</span>`
                        : `<button class="btn btn-danger btn-sm" style="margin-left: auto;" onclick="voidFullSale('${s.id}')">Void Sale</button>`;

                    const div = document.createElement('div');
                    div.className = rowClass;
                    div.style.cursor = isFullyVoided ? 'default' : 'pointer';
                    div.onclick = () => { if(!isFullyVoided) openTransactionDetails(s.id); };
                    div.innerHTML = `
                        <div class="inv-details">
                            <div style="font-weight:bold">${time} ${hasVoidedItems ? '(Items Voided)' : ''}</div>
                            <div class="text-small truncate-text" title="${itemSummary}">${itemSummary} • ${s.method}</div>
                        </div>
                        <div style="font-weight:bold; color:${isFullyVoided ? 'var(--gray)' : 'var(--primary)'}">
                            ${formatMoney(netTotal)}
                        </div>
                        ${voidAction}
                    `;
                    list.appendChild(div);
                });
            }
        }

        function exportCSV() {
            if (!isAppUnlocked()) { showToast("License required to export data"); return; }
            if(state.sales.length === 0) { showToast("No data to export"); return; }

            let csvContent = "data:text/csv;charset=utf-8,";
            csvContent += "Date,Time,Total,Cost,Profit,Method,Voided,Items Sold\n";

            state.sales.forEach(s => {
                const d = new Date(s.timestamp);
              
                // 5️⃣ Use stored profit if available
                let profit = s.profit;
                if (profit === undefined) {
                    let sProfit = 0;
                    let sTotal = 0;
                    s.items.forEach(i => {
                        if (typeof i.voided === 'undefined') i.voided = false;
                        if(!i.voided) {
                            sTotal += (i.price * i.qty);
                            sProfit += (i.price * i.qty) - (i.cost * i.qty);
                        }
                    });
                    profit = sProfit;
                }

                let netTotal = 0;
                let netCost = 0;
                let activeItems = [];
              
                s.items.forEach(i => {
                    if (typeof i.voided === 'undefined') i.voided = false;
                    if(!i.voided) {
                        netTotal += (i.price * i.qty);
                        netCost += (i.cost * i.qty);
                        activeItems.push(i);
                    }
                });

                const itemSummary = activeItems.map(i => `${i.name} x${i.qty}`).slice(0, 3).join('; ');
                if(activeItems.length > 3) itemSummary += ` +${activeItems.length - 3} more`;

                csvContent += `${d.toLocaleDateString()},${d.toLocaleTimeString()},${netTotal},${netCost},${profit},${s.method},${s.voided ? 'Yes' : 'No'},"${itemSummary}"\n`;
            });

            const encodedUri = encodeURI(csvContent);
            const link = document.createElement("a");
            link.setAttribute("href", encodedUri);
            link.setAttribute("download", "sales_export.csv");
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
        }

        // --- 7️⃣ TOTALS RECALCULATION (FOR IMMUTABILITY) ---
        function recalcTotalsFromHistory() {
            let gross = 0;
            let profit = 0;
            let count = 0;

            state.sales.forEach(s => {
                if (s.voided) return;
               
                // 9️⃣ Safety Guard: Handle missing 'profit' field from old transactions
                if (s.profit !== undefined) {
                    gross += s.total;
                    profit += s.profit;
                    count++;
                } else {
                    // Calculate on the fly if missing
                    let sProfit = 0;
                    s.items.forEach(i => {
                        if (!i.voided) {
                            gross += (i.price * i.qty);
                            sProfit += ((i.price - i.cost) * i.qty);
                        }
                    });
                    profit += sProfit;
                    count++;
                }
            });

            // Only overwrite if history is different from stored totals (prevents unnecessary writes)
            if (state.totals.gross !== gross || state.totals.profit !== profit || state.totals.transactions !== count) {
                state.totals = { gross, profit, transactions: count };
                saveState();
            }
        }

        // --- HELPERS ---
        function getTransactionSummary(items) {
            if (!items || items.length === 0) return "No items";
            const displayLimit = 3;
            const limitedItems = items.slice(0, displayLimit);
            const itemStrings = limitedItems.map(i => `${i.name} x${i.qty}`);
            let result = itemStrings.join(", ");
            if (items.length > displayLimit) {
                const remaining = items.length - displayLimit;
                result += ` +${remaining} more`;
            }
            return result;
        }
